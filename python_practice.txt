#*Q37.* Standardize the values in the Gender column so that M, Male, and MALE are represented consistently. Do the same for female values.

# Reload original Gender column to discard messed up values
original_df = pd.read_csv('bank_customer_messy_practice.csv')
df['Gender'] = original_df['Gender']

# Standardize Gender cleanly using mapping
gender_map = {
    'Female': 'Female', 'female ': 'Female', 'female': 'Female', 'FEMALE': 'Female', 'F': 'Female',
    'Male': 'Male', 'Male ': 'Male', 'male': 'Male', 'MALE': 'Male', 'M': 'Male'
}
df['Gender'] = df['Gender'].map(gender_map)
